# Runtime dependencies for the vendored fashn_vton package.
# NOTE: `-e .` is intentionally NOT used here — the ZeroGPU build runs
# `pip install -r requirements.txt` in a base stage BEFORE the app source is
# copied in, so an editable install would fail. The package is imported via a
# sys.path shim in app.py instead.

# ZeroGPU only supports torch 2.8.0 / 2.9.1 / 2.10.0 / 2.11.0 (CUDA is
# provided by the ZeroGPU runtime itself, so plain PyPI wheels are correct).
torch==2.9.1
torchvision==0.24.1

onnxruntime-gpu==1.23.2
safetensors>=0.3.0
# gradio[oauth]==4.44.1 (installed by the ZeroGPU runtime) imports
# huggingface_hub.HfFolder, which was REMOVED in huggingface_hub 1.x.
# Pin below 1.0 to keep the runtime importable.
huggingface_hub>=0.20.0,<1.0
# fashn-human-parser requires transformers>=4.30. Keep it on 4.x: transformers 5.x
# depends on huggingface_hub>=1.0, which would conflict with the pin above.
transformers>=4.30,<5
pillow>=9.0.0
numpy>=1.21.0
opencv-python>=4.5.0
tqdm>=4.65.0
einops>=0.6.0
matplotlib>=3.5.0
fashn-human-parser>=0.1.1

# The ZeroGPU runtime installs gradio[oauth]==4.44.1 and spaces==0.51.1 itself,
# so pip must NOT upgrade gradio (5.x breaks under ZeroGPU). This forces a
# known Pydantic incompatibility on launch (see the comment near the top of
# app.py for the monkey-patch): Pydantic >= 2.11 emits boolean schema nodes
# that gradio_client 1.3.0 chokes on with
#   TypeError: argument of type 'bool' is not iterable
# Pinning pydantic < 2.11 is the community-verified fix (ShortGPT, AllTalk,
# TRELLIS) and matches the Gradio 4.44.1-era dependency set.
gradio==4.44.1
pydantic>=2.6,<2.11
# Gradio 4.44.1 renders its page template via Starlette's OLD
# TemplateResponse(name, context) signature. Starlette 1.0 removed that
# signature (context dict gets passed as the template name → Jinja2
# "TypeError: unhashable type: 'dict'"). Pin the last compatible versions.
fastapi==0.114.2
starlette==0.38.6
jinja2==3.1.4
